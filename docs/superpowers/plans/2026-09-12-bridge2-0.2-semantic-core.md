# BRIDGE² 0.2 Semantic Core Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Introduce a provider-neutral BRIDGE IR, structured diagnostics, and adapter ABI underneath the existing public API while preserving all 15 v0.1.3 generated artifacts byte-for-byte.

**Architecture:** Keep `SkillSpec`, CLI commands, MCP tools, and `daube_bridge.compiler` imports as compatibility surfaces. Internally parse `SkillSpec → BridgeIR`, plan/emit through built-in target adapters, then merge artifacts through a deterministic pipeline. Capability resolution is deliberately deferred to 0.3; 0.2 establishes the semantic and adapter seams without changing observable output.

**Tech Stack:** Python 3.11+, dataclasses, typing `Protocol`, PyYAML, FastMCP 4+, pytest 8+, Ruff, Hatchling.

**Spec:** `docs/superpowers/specs/2026-09-12-bridge2-ultimate-architecture-design.md`

## Global Constraints

- Preserve the existing `skill.yaml` input contract and `SkillSpec.from_dict()` validation behavior.
- Preserve `compile_spec(spec, endpoint) -> dict[str, str]` and the existing CLI/MCP tool behavior.
- Preserve exactly 15 v0.1.3 artifacts for `examples/skill.yaml` before intentional provider-format changes.
- Provider names/syntax may exist in adapters, never in canonical IR.
- Compilation remains offline and deterministic by default.
- No capability resolver, Studio, Tasks, Apps, OAuth, catalog, sandbox, or package split in 0.2.
- Base dependencies remain `fastmcp` + `PyYAML`; do not add runtime dependencies.
- Every behavior change follows RED → GREEN → full regression → commit.

```python
def slugify(value: str) -> str:
```

```python
from dataclasses import asdict, dataclass
from enum import StrEnum

class Severity(StrEnum):
    INFO = "info"
    WARNING = "warning"
    ERROR = "error"

@dataclass(frozen=True, slots=True)
class SourceLocation:
    file: str | None = None
    path: str | None = None
    line: int | None = None

@dataclass(frozen=True, slots=True)
class Diagnostic:
    code: str
    severity: Severity
    message: str
    source: SourceLocation | None = None
    target: str | None = None
    capability: str | None = None
    suggestions: tuple[str, ...] = ()

    def to_dict(self) -> dict[str, object]:
        data = asdict(self)
        data["severity"] = self.severity.value
        return data
```

```python
# src/daube_bridge/adapters/base.py
from dataclasses import dataclass
from typing import Any, Protocol
from daube_bridge.core.diagnostics import Diagnostic
from daube_bridge.core.ir import BridgeIR

ArtifactMap = dict[str, str]

@dataclass(frozen=True, slots=True)
class TargetPlan:
    target: str
    ir: BridgeIR
    endpoint: str
    metadata: dict[str, Any] | None = None

class TargetAdapter(Protocol):
    target: str
    def probe(self) -> dict[str, Any]: ...
    def plan(self, ir: BridgeIR, endpoint: str) -> TargetPlan: ...
    def emit(self, plan: TargetPlan) -> ArtifactMap: ...
    def verify(self, artifacts: ArtifactMap) -> tuple[Diagnostic, ...]: ...
```

Implement `shared.py` with the legacy rendering contract preserved exactly:

```python
import json
from daube_bridge.core.ir import BridgeIR


def pretty_json(value: object) -> str:
    return json.dumps(value, indent=2, ensure_ascii=False) + "\n"


def skill_markdown(ir: BridgeIR) -> str:
    skill = ir.skill
    return (
        f"---\nname: {skill.slug}\ndescription: {skill.description}\n---\n\n"
        f"# {skill.name}\n\n{skill.description}\n\n"
        "Use the D'AUBE Bridge MCP server when these capabilities are relevant.\n"
    )


def function_tools(ir: BridgeIR) -> list[dict[str, object]]:
    return [
        {"type": "function", "function": {
            "name": tool.name, "description": tool.description,
            "parameters": tool.input_schema,
        }}
        for tool in ir.tools
    ]


def mcp_http_config(ir: BridgeIR, endpoint: str) -> dict[str, object]:
    return {"mcpServers": {ir.skill.slug: {"type": "http", "url": endpoint}}}
```

```python
@dataclass(frozen=True, slots=True)
class CompilationResult:
    artifacts: dict[str, str]
    diagnostics: tuple[Diagnostic, ...]

def compile_ir(ir: BridgeIR, endpoint: str) -> CompilationResult:
    artifacts = {"bridge/manifest.json": pretty_json(bridge_manifest(ir, endpoint, TARGETS))}
    diagnostics: list[Diagnostic] = []
    for adapter in BUILTIN_ADAPTERS:
        plan = adapter.plan(ir, endpoint)
        emitted = adapter.emit(plan)
        duplicate = artifacts.keys() & emitted.keys()
        if duplicate:
            raise ValueError(f"Duplicate artifact paths: {', '.join(sorted(duplicate))}")
        diagnostics.extend(adapter.verify(emitted))
        artifacts.update(emitted)
    return CompilationResult(artifacts, tuple(diagnostics))
```

```python
from __future__ import annotations

import hashlib
import tomllib
import zipfile
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
FIXED_TIME = (2020, 1, 1, 0, 0, 0)

def project_version() -> str:
    data = tomllib.loads((ROOT / "pyproject.toml").read_text(encoding="utf-8"))
    return str(data["project"]["version"])

def bundle_files() -> dict[str, Path]:
    files = {
        "manifest.json": ROOT / "mcpb" / "manifest.json",
        "pyproject.toml": ROOT / "mcpb" / "pyproject.toml",
        "src/server.py": ROOT / "mcpb" / "src" / "server.py",
    }
    source_root = ROOT / "src"
    for source in sorted((source_root / "daube_bridge").rglob("*.py")):
        files[source.relative_to(source_root).as_posix()] = source
    return files
```
```python
def build_bundle() -> tuple[Path, str]:
    version = project_version()
    out = ROOT / "dist" / f"daube-agent-bridge-{version}.mcpb"
    out.parent.mkdir(parents=True, exist_ok=True)
    with zipfile.ZipFile(out, "w", zipfile.ZIP_DEFLATED, compresslevel=9) as archive:
        for name, source in sorted(bundle_files().items()):
            info = zipfile.ZipInfo(name, FIXED_TIME)
            info.compress_type = zipfile.ZIP_DEFLATED
            info.external_attr = 0o644 << 16
            archive.writestr(info, source.read_bytes())
    digest = hashlib.sha256(out.read_bytes()).hexdigest()
    return out, digest

def main() -> int:
    out, digest = build_bundle()
    print(f"MCPB={out}")
    print(f"SHA256={digest}")
    return 0

if __name__ == "__main__":
    raise SystemExit(main())
```

```python
def root_version() -> str:
    data = tomllib.loads((ROOT / "pyproject.toml").read_text(encoding="utf-8"))
    return str(data["project"]["version"])

def _write_json(path: Path, data: dict) -> None:
    path.write_text(
        json.dumps(data, indent=2, ensure_ascii=False) + "\n",
        encoding="utf-8",
    )

def sync_versions() -> str:
    version = root_version()
    for relative in JSON_VERSION_FILES:
        path = ROOT / relative
        data = json.loads(path.read_text(encoding="utf-8"))
        data["version"] = version
        if relative == Path("server.json"):
            data["packages"][0]["identifier"] = (
                "https://github.com/daubesonntag-dotcom/daube-agent-bridge/"
                f"releases/download/v{version}/daube-agent-bridge-{version}.mcpb"
            )
        _write_json(path, data)

    path = ROOT / "mcpb" / "pyproject.toml"
    text = path.read_text(encoding="utf-8")
    text = re.sub(r'(?m)^version = "[^"]+"$', f'version = "{version}"', text, count=1)
    path.write_text(text, encoding="utf-8")
    return version
```

```python
def main() -> int:
```
